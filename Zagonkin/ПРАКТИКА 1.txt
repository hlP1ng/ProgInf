#ifndef BATTERY_H
#define BATTERY_H

#include <string>

class Battery {
private:
    int cell_count; //int - целое число
    double voltage; //double - дробное число
    double capacity_mAh;

public:
    Battery(); //конструктор
    Battery(int cells, double initial_voltage, double capacity); //принимает параметры

    double get_percentage() const;
    bool is_low() const;
    void set_voltage(double new_voltage);
    void print_info() const;
    std::string get_status() const;

    // Эти три метода нужны, чтобы окно могло прочитать данные
    int get_cells() const { return cell_count; }
    double get_voltage() const { return voltage; }
    double get_capacity() const { return capacity_mAh; }
};

#endif // BATTERY_H


battery.cpp
#include "battery.h"
#include <iostream>

Battery::Battery() //дефолтный конструктор
    : cell_count(4),
    voltage(0.0),
    capacity_mAh(5000.0) {
}

Battery::Battery(int cells, double initial_voltage, double capacity)
    : cell_count(cells),
    voltage(initial_voltage),
    capacity_mAh(capacity) {
    if (cell_count < 1) {
        cell_count = 1;
    }
    if (capacity_mAh < 0.0) {
        capacity_mAh = 0.0;
    }
}

double Battery::get_percentage() const {
    const double min_voltage = 3.0 * cell_count;
    const double max_voltage = 4.2 * cell_count;

    double result = (voltage - min_voltage) / (max_voltage - min_voltage) * 100.0;

    if (result < 0.0) {
        result = 0.0;
    }
    if (result > 100.0) {
        result = 100.0;
    }
    return result;
}

bool Battery::is_low() const {
    return get_percentage() < 20.0;
}

void Battery::set_voltage(double new_voltage) {
    if (new_voltage < 0.0) {
        std::cout << "Invalid voltage\n";
        return;
    }
    voltage = new_voltage;
}

void Battery::print_info() const {
    std::cout << "Cells: " << cell_count << '\n';
    std::cout << "Voltage: " << voltage << " V\n";
    std::cout << "Capacity: " << capacity_mAh << " mAh\n";
    std::cout << "Charge: " << get_percentage() << "%\n";
    std::cout << "Status: " << get_status() << '\n';
}

std::string Battery::get_status() const {
    double p = get_percentage();
    if (p < 10.0) return "CRITICAL";
    if (p < 20.0) return "LOW";
    if (p <= 80.0) return "NORMAL";
    return "FULL";
}
